# routing-bench programs (ADR-0027 §3, #768).
#
# The arm modules installed beside routing-bench are found here, not listed:
# a module joins the harness by shipping helper/<module>-bench-arms.cc, which
# defines ns3::bench::RegisterArms_<module>() (hyphens become underscores).
# CMake generates arms-generated.cc, which calls the stock arms' registration
# and then each present, enabled module's, so registration is an explicit
# call visible in the build -- never a static initialiser, which a link with
# --as-needed could drop silently.
#
# The link list is built from variables rather than written out: ns-3 scans
# module CMakeLists for library tokens to resolve dependencies, and the arm
# modules depend on routing-bench, not the reverse.

set(rb_arm_modules)
file(GLOB rb_arm_files ${PROJECT_SOURCE_DIR}/contrib/*/helper/*-bench-arms.cc)
foreach(f ${rb_arm_files})
  get_filename_component(rb_dir ${f} DIRECTORY)
  get_filename_component(rb_dir ${rb_dir} DIRECTORY)
  get_filename_component(rb_mod ${rb_dir} NAME)
  if(rb_mod IN_LIST ns3-all-enabled-modules)
    list(APPEND rb_arm_modules ${rb_mod})
  endif()
endforeach()
list(SORT rb_arm_modules)
list(REMOVE_DUPLICATES rb_arm_modules)

set(rb_decls "")
set(rb_calls "")
set(rb_libs)
foreach(m ${rb_arm_modules})
  string(REPLACE "-" "_" rb_fn ${m})
  string(APPEND rb_decls "void RegisterArms_${rb_fn}();\n")
  string(APPEND rb_calls "    RegisterArms_${rb_fn}();\n")
  set(rb_libvar "lib${m}")
  list(APPEND rb_libs ${${rb_libvar}})
endforeach()
message(STATUS "routing-bench: arm modules found: ${rb_arm_modules}")

set(rb_generated ${CMAKE_CURRENT_BINARY_DIR}/arms-generated.cc)
file(WRITE ${rb_generated}.in
"// Generated by routing-bench/examples/CMakeLists.txt -- do not edit.
namespace ns3 {
namespace bench {
void RegisterStockArms();
${rb_decls}
void RegisterInstalledArms() {
    RegisterStockArms();
${rb_calls}}
}  // namespace bench
}  // namespace ns3
")
configure_file(${rb_generated}.in ${rb_generated} COPYONLY)

foreach(rb_prog routing-compare anthocnet-compare)
  build_lib_example(
    NAME ${rb_prog}
    SOURCE_FILES ${rb_prog}.cc ${rb_generated}
    LIBRARIES_TO_LINK
      ${librouting-bench}
      ${libcore}
      ${libnetwork}
      ${libinternet}
      ${libwifi}
      ${libmobility}
      ${libapplications}
      ${libflow-monitor}
      ${libenergy}
      ${libaodv}
      ${libolsr}
      ${libdsdv}
      ${libmesh}
      ${rb_libs}
  )
endforeach()
